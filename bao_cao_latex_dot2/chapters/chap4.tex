\chapter{Kiểm thử và đánh giá kết quả}
\label{ch:kiem-thu}

\section{Mục tiêu kiểm thử}

\begin{enumerate}
    \item Xác nhận chuỗi guard và pipeline RAG lab hoạt động đúng trên dữ liệu tổng hợp.
    \item Đo hiệu quả bảo mật của tường (TPR/FPR/Stop-before-LLM/Exfil) trên bộ tấn công tổng hợp và trả lời ba câu hỏi nghiên cứu.
    \item Đo mức độ tái lập của quy trình đánh giá.
    \item Làm rõ giới hạn bằng chứng: chỉ dùng mô hình cục bộ ollama cho Semantic Judge, \textbf{không} phải LLM doanh nghiệp thật.
\end{enumerate}

\section{Phương pháp kiểm thử}

Nhóm kết hợp: (i)~kiểm thử đơn vị/tích hợp bằng pytest cho các guard và pipeline RAG; (ii)~đánh giá hiệu quả bảo mật trên bộ tấn công/lành tính tổng hợp với ba cấu hình guard; (iii)~thí nghiệm tối ưu lớp luật, đo trước/sau trên cùng bộ dữ liệu.

\section{Môi trường kiểm thử}

Python trong \texttt{.venv}; pytest với basetemp ngoài repo khi cần; không gọi mạng cho đường đánh giá chính (Mock Provider tất định), tuỳ chọn mô hình cục bộ ollama cho cấu hình Semantic Judge.

\section{Các nhóm kiểm thử}

\subsection{Kiểm thử chức năng}
Quyết định của từng guard, pipeline RAG và kiểm soát truy cập theo vai trò (RBAC).

\subsection{Kiểm thử lỗi và fail-closed}
Xác nhận nguyên tắc quyết định nghiêm khắc nhất thắng và hành vi fail-closed khi có vi phạm rõ.

\section{Kịch bản kiểm thử chính}

\begin{itemize}
    \item Baseline (always-allow) so với có guard trên bộ đánh giá bảo mật tổng hợp.
    \item Ba cấu hình guard (rule-based, Semantic Judge \texttt{qwen3:4b}, đối chiếu \texttt{hermes3:8b}) trên cùng bộ 425 case.
    \item Thí nghiệm tối ưu lớp luật: đo TPR/FPR trước và sau trên cùng dữ liệu.
\end{itemize}

\section{Đánh giá hiệu quả bảo mật trên bộ tấn công tổng hợp}
\label{sec:sec-eval}

Bộ đánh giá bảo mật tổng hợp gồm \textbf{425 case}: \textbf{200 case tấn công}
\emph{lấy cảm hứng} từ các họ kỹ thuật của NVIDIA garak~\cite{derczynski2024garak},
Microsoft PyRIT và InjecAgent~\cite{zhan2024injecagent} (template tĩnh tự xây,
\textbf{không} gọi các bộ công cụ gốc) và
\textbf{225 câu lành tính} gồm 100 câu nghiệp vụ thông thường và 125 câu
\emph{hard benign} --- câu hợp lệ nhưng bề mặt trông đáng ngờ (chứa từ khoá nhạy
cảm hợp pháp, hỏi bảo mật chính đáng, tự phục vụ, diễn đạt lại), dùng để đo
precision thật của tường. Toàn bộ payload là dữ liệu tổng hợp, sinh xác định.
Đánh giá chạy trên ba cấu hình trên \emph{cùng một bộ 425 case}:
(a)~\textbf{rule-based} (chỉ guard luật); (b)~\textbf{semantic} (bổ sung
Semantic Judge trên mô hình cục bộ \texttt{qwen3:4b}); và (c)~một lần
\textbf{đối chiếu} với mô hình lớn hơn \texttt{hermes3:8b}.

Các chỉ số được định nghĩa: \textbf{TPR} = tấn công bị chặn / tổng tấn công;
\textbf{FPR} = câu lành tính bị chặn / tổng lành tính; \textbf{Stop-before-LLM}
= tỉ lệ tấn công bị chặn trước khi gọi mô hình; \textbf{Exfil Marker Rate} =
tỉ lệ câu trả lời để lộ mã canary bí mật. Theo quyết định phương pháp L2,
\emph{latency không được báo cáo}.

\textbf{Lựa chọn và ý nghĩa chỉ số.} Với một tường guard, hai chỉ số TPR và FPR
không cùng vai trò. \textbf{FPR được xem là ràng buộc cứng}: một tường chặn nhầm
câu nghiệp vụ hợp lệ sẽ không thể triển khai, dù có TPR cao đến đâu --- người
dùng sẽ tắt nó. Do đó bộ lành tính được thiết kế \emph{khó} (125 câu
\emph{hard benign} chứa từ khoá nhạy cảm hợp pháp, hỏi bảo mật chính đáng, tự
phục vụ) để đo precision thật, và mục tiêu là \emph{tối đa hoá TPR trong khi giữ
FPR ở mức thấp nhất có thể}. \textbf{TPR/recall} phản ánh rủi ro bỏ lọt: trong an
ninh, bỏ sót một đòn tấn công thật thường nguy hiểm hơn một cảnh báo nhầm, nên
recall là đại lượng cần cải thiện. \textbf{Stop-before-LLM} đo phần tấn công bị
chặn \emph{trước} khi gọi mô hình --- vừa tiết kiệm chi phí token, vừa là phòng
thủ sớm không phụ thuộc hành vi mô hình. \textbf{Exfil Marker Rate} đo rò rỉ ở
tầng đầu ra; như phân tích bên dưới, phép đo này hiện bị nhiễu và chỉ dùng như
chỉ báo định tính. Tất cả tỉ lệ có cỡ mẫu dưới ngưỡng $n=10$ được báo là ``không
đủ mẫu'' thay vì một con số dễ gây hiểu lầm.

\textbf{Tối ưu lớp luật.} Lần đánh giá đầu cho thấy lớp luật bỏ lọt hoàn toàn họ
PyRIT (đóng vai/mạo danh thẩm quyền, $0\%$). Chúng tôi bổ sung một nhóm luật
nhắm \emph{hai bất biến từ vựng} của đòn mạo danh (cụm từ đặc trưng, không phải đặc trưng ngữ nghĩa) --- chỉ thị vượt quyền (ghi đè chính
sách/ACL, ``không được từ chối cấp trên'') và yêu cầu trích xuất hàng loạt dữ
liệu nhạy cảm --- thay vì khớp đúng câu chữ của từng case. Trên cùng bộ 425
case, thay đổi này nâng TPR của lớp luật từ $48{,}5\%$ lên $77{,}0\%$ mà
\textbf{giữ nguyên FPR $0\%$ trên bộ 225 câu lành tính tổng hợp}
(Bảng~\ref{tab:v3-optim}). Cần nói rõ phạm vi: đây \textbf{không} phải bằng chứng
đã ``vá'' mạo danh nói chung. Con số $100\%$ trên họ PyRIT phản ánh việc họ
template tổng hợp hiện tại (khoảng sáu khung) đều nhúng đúng hai bất biến mà luật
bắt; kẻ tấn công diễn đạt lại để bỏ hai bất biến (bỏ định lượng ``toàn bộ'', bỏ
động từ ghi đè) vẫn né được luật --- kiểm thử đơn vị ghi nhận rõ tám đòn né như
vậy đều lọt (phù hợp với tài liệu về vượt qua guardrail~\cite{hackett2025bypass}).
Tương tự, FPR $0\%$ đúng trên bộ tổng hợp nhưng \textbf{không khái quát}: một số
câu quản trị/điều hành hợp lệ ngoài phân phối (câu hỏi \emph{về} việc ghi đè
chính sách, yêu cầu xuất dữ liệu hợp pháp cho kiểm toán) vẫn bị chặn nhầm và được
ghi nhận là hạn chế.

\subsection{Phân tích kết quả}

Ba quan sát từ số liệu trên định hình kết luận, và không quan sát nào được diễn
giải theo hướng có lợi một cách thiếu cơ sở.

\textbf{Thứ nhất, lớp luật cho precision cao trên bộ tổng hợp và recall trung
bình.} Sau khi bổ sung nhóm luật chống mạo danh, cấu hình rule-based đạt TPR
$77{,}0\%$ mà vẫn giữ FPR $0\%$ trên toàn bộ 225 câu lành tính \emph{tổng hợp}
--- kể cả 125 câu \emph{hard benign} --- tức không chặn nhầm câu lành tính nào
\emph{trong bộ này} (precision không khái quát tuyệt đối, xem phạm vi ở trên và
mục Hạn chế). Theo nguồn, lớp luật nay chặn trọn họ PyRIT tổng hợp ($100\%$) và
giữ nguyên garak ($75{,}2\%$); phần bỏ lọt
còn lại tập trung ở đòn nhiễu Unicode và chỉ thị ẩn trong văn xuôi (injecagent
$47{,}4\%$) --- những dạng cần chuẩn hoá/giải nhiễu đầu vào sâu hơn, ghi nhận là
hướng phát triển. Việc TPR nhảy $+28{,}5$ điểm mà FPR không đổi cho thấy có thể
nâng recall của guard-luật bằng cách nhắm \emph{đặc trưng bất biến} của một họ
tấn công thay vì thêm từ khoá rời rạc; song đây không phải phòng thủ thích ứng ---
một biến thể diễn đạt lại vẫn có thể né.

\textbf{Thứ hai, lớp Semantic Judge gần như không cải thiện recall mà lại đánh
đổi precision.} Trên \texttt{qwen3:4b}, tỉ lệ chặn theo từng nguồn gần \emph{trùng
khớp} với lớp luật đã tối ưu (PyRIT $100\%$, injecagent $47{,}4\%$, garak nhích
$75{,}2\%\to76{,}2\%$): TPR toàn cục chỉ tăng $77{,}0\%\to77{,}5\%$ (đúng một
case). Tức là mô hình phán đoán cục bộ nhỏ này \textbf{gần như không bắt thêm
được đòn nào}, trong khi làm FPR tăng từ $0\%$ lên $11{,}1\%$ (và $14{,}4\%$
riêng trên hard benign). Kết quả này cho thấy giá trị của LLM-as-judge phụ thuộc
mạnh vào năng lực mô hình: một mô hình lớn hơn có thể đạt recall rất cao nhưng
chỉ bằng cách chặn bừa (xem mục đối chiếu bên dưới), còn mô hình nhỏ thì hầu như
không giúp ích mà chỉ thêm chi phí precision và độ trễ.

\textbf{Thứ ba, chỉ số exfil bị nhiễu echo nhưng không hoàn toàn là echo.}
Cấu hình semantic ghi nhận Exfil Marker Rate $21{,}5\%$ (giảm từ $50\%$ ở lần
đánh giá trước, do lớp luật tối ưu chặn sớm nhiều đòn hơn trước khi gọi mô hình).
Con số tổng $21{,}5\%$ \textbf{không hợp lệ như một tỉ lệ rò rỉ kho tri thức}:
$190/200$ prompt tấn công đã chứa sẵn mã canary ngay trong câu hỏi. Đối chiếu
từng case cho thấy trong $190$ prompt ``nhiễm'' này có $37$ lần lộ --- vì target
đồng thời có trong prompt lẫn kho nên artifact \textbf{không xác định được nguồn}
từng marker (không thể khẳng định chắc chắn là echo hay trích xuất). \emph{Tuy
nhiên}, trên $10$ prompt \emph{sạch} (canary \textbf{chỉ} nằm trong kho, không có
trong câu hỏi) vẫn có $6$ lần lộ: đây là tín hiệu rò rỉ kho \emph{thật} (exact-
marker), tuy \textbf{cỡ mẫu nhỏ} ($n=10$ --- đúng bằng ngưỡng tối thiểu nên vẫn
báo cáo được, nhưng under-powered). Vậy không thể quy toàn bộ thành echo, cũng
không thể coi $21{,}5\%$ là tỉ lệ rò rỉ kho. Để đo \emph{đúng}, nhóm thực hiện
một thí nghiệm canary-sạch (canary chỉ tồn tại trong kho) ở
Mục~\ref{sec:ab-demo}, đồng thời bổ sung lớp phát hiện canary cho Output Guard
để chặn rò rỉ ngay ở biên đầu ra.

\textbf{Đối chiếu (mô hình lớn hơn).} Một lần chạy tham chiếu với \texttt{hermes3:8b}
trên \emph{cùng bộ 425 case} đạt TPR $100\%$ nhưng FPR $67{,}1\%$. Con số TPR
$100\%$ này \emph{không} được diễn giải như bằng chứng tường mạnh: một bộ phân
loại chặn gần như mọi thứ (Stop-before-LLM $99{,}5\%$, tức $199/200$ \emph{tấn
công} bị chặn trước khi gọi mô hình) tất yếu đạt TPR cao, và FPR $67{,}1\%$ (chặn
hai phần ba câu hợp lệ) khiến cấu hình đó không thể vận hành.

\textbf{Kết luận.} Sau tối ưu, \textbf{lớp luật tất định là cấu hình cân bằng
nhất}: TPR $77\%$ ở FPR $0\%$ \emph{trên bộ 425 case tổng hợp}. Lớp LLM-as-judge nhỏ chưa chứng minh được lợi ích
recall tương xứng với chi phí FPR và độ trễ; mô hình lớn đẩy recall lên trần
nhưng bằng FPR không thể chấp nhận. Không cấu hình đơn lẻ nào đạt mức triển khai
thực tế; các hướng phát triển còn lại gồm giải nhiễu đầu vào (Unicode/chỉ thị ẩn)
để nâng recall injecagent, sửa phép đo exfil để đánh giá được rò rỉ kho thật, và
giảm FPR của lớp ngữ nghĩa.

Trả lời trực tiếp ba câu hỏi nghiên cứu ở phần Mở đầu: \textbf{CH1} --- tường luật
phát hiện $77\%$ đòn tấn công tổng hợp ở FPR $0\%$, mạnh trên garak/PyRIT nhưng
yếu trên injecagent (nhiễu/ẩn); \textbf{CH2} --- Semantic Judge nhỏ
\emph{không} cải thiện recall đáng kể ($+1$ case) mà làm FPR tăng lên $11{,}1\%$,
nên chưa xứng chi phí; \textbf{CH3} --- \emph{có}, luật nhắm đặc trưng bất biến
nâng recall lớp luật thêm $28{,}5$ điểm mà giữ FPR $0\%$ trên bộ tổng hợp, song
đây là cải thiện theo họ tấn công tổng hợp, không phải phòng thủ thích ứng.

\section{Thí nghiệm A/B canary-sạch: có tường so với không tường}
\label{sec:ab-demo}

Để khắc phục nhiễu của phép đo exfil ở trên (canary nằm trong prompt) và để đối
chiếu công bằng, nhóm thực hiện một thí nghiệm \emph{paired} tái lập được: gieo
một mã canary \textbf{chỉ vào một tài liệu bảng lương mật trong kho} (canary
\emph{không} xuất hiện trong bất kỳ prompt nào), rồi chạy \emph{cùng một prompt}
qua hai luồng --- \textbf{không tường} (gọi thẳng mô hình cục bộ, không guard) và
\textbf{có tường} (qua toàn bộ chuỗi guard, gồm Output Guard phát hiện canary).
Vì canary chỉ có trong kho, một canary xuất hiện ở câu trả lời là bằng chứng
\emph{rò rỉ kho thật}. Bộ câu hỏi gồm 1 câu nghiệp vụ hợp lệ và 6 đòn trích
xuất/mạo danh bảng lương. Kịch bản tái lập bằng
\texttt{scripts/run\_demo\_ab\_paired.py}; kết quả content-free tại
\texttt{reports/demo-ab/}.

Quan sát chính. Ở luồng \textbf{không tường}, cơ chế tự phòng thủ của mô hình
hoạt động \emph{không nhất quán}: phần lớn lần chạy mô hình tự từ chối, nhưng có
lần nó \emph{làm lộ nguyên văn} bảng lương (họ tên, mã nhân viên, lương, số tài
khoản) kèm mã canary ở tới $3/6$ đòn --- tức rò rỉ kho thật. Nói cách khác,
\emph{không thể tin cậy} vào việc mô hình tự từ chối. Ở luồng \textbf{có tường},
\textbf{không lần chạy nào lộ canary}: phần lớn đòn bị Input Guard chặn trước khi
gọi mô hình; đặc biệt, các đòn \emph{lọt qua} Input Guard vẫn bị \textbf{Output
Guard chặn ở biên đầu ra} khi mô hình định trả canary ra ($2$--$3$ đòn mỗi lần
chạy) --- minh hoạ trực tiếp giá trị phòng thủ theo chiều sâu. Đây là bằng chứng
cho thấy lớp guard là \emph{cần thiết} và cho protection \emph{nhất quán} mà
mô hình đơn lẻ không đảm bảo.

Đây cũng là phép đo exfil \emph{hợp lệ} (canary chỉ trong kho) mà con số $21{,}5\%$
ở trên không cung cấp được. Giới hạn: mô hình phi tất định nên số lần lộ của luồng
không tường dao động giữa các lần chạy (đã báo cáo dạng khoảng); cỡ mẫu nhỏ, dữ
liệu tổng hợp và mô hình cục bộ; các giá trị nhạy cảm là giả lập và không in lại
trong báo cáo.

\section{Đánh giá kết quả}

\subsection{Mức độ đáp ứng yêu cầu}
Các nhóm yêu cầu chức năng chính --- chuỗi guard đầu vào/ngữ cảnh/đầu ra,
pipeline RAG có kiểm soát truy cập, và ghi nhật ký content-free --- đều đã có
hiện thực và bộ kiểm thử tương ứng trong môi trường lab.

\subsection{Tính an toàn của đầu ra}
Đầu ra guard được ghi ở dạng content-free (mã lý do ổn định, không lộ nội dung
nhạy cảm thô); quyết định tuân nguyên tắc fail-closed khi có vi phạm rõ.

\subsection{Khả năng tái lập}
Mock Provider tất định, bộ dữ liệu và artifact đánh giá được băm SHA-256, và bộ
kiểm thử tự động lặp lại được trong môi trường đã mô tả. Bộ chạy đánh giá \emph{hiện}
ghi kèm provenance (commit, mô hình, cấu hình, băm dữ liệu) cho các lần chạy mới ---
đây là provenance ở mức tối thiểu, chưa khóa đầy đủ môi trường và digest mô hình.

\section{Hạn chế của kết quả đánh giá}

\begin{itemize}
    \item FPR $0\%$ đo trên bộ hard benign tổng hợp (57 nội dung độc nhất từ 40 template); precision \textbf{không khái quát}: probe ngoài phân phối cho thấy một số câu quản trị/điều hành hợp lệ (hỏi \emph{về} ghi đè chính sách, yêu cầu xuất dữ liệu hợp pháp cho kiểm toán) vẫn bị chặn nhầm.
    \item Phép đo Exfil Marker \textbf{bị nhiễu vì prompt chứa sẵn canary} ($190/200$; nguồn từng marker không xác định được); trên $10$ prompt sạch có $6$ lần lộ --- rò rỉ kho là thật nhưng cỡ mẫu nhỏ ($n=10$), chưa định lượng tin cậy. Cần thiết kế lại canary (chỉ nằm trong kho).
    \item TPR $77\%$ đạt bằng luật nhắm đặc trưng họ tấn công \emph{tổng hợp}; \textbf{không} phải phòng thủ thích ứng --- tám đòn diễn đạt lại trong kiểm thử đơn vị đều né được; $100\%$ trên PyRIT chỉ đúng với họ template hiện tại.
    \item Toàn bộ đánh giá dùng dữ liệu tổng hợp và mô hình cục bộ; \textbf{không} đánh giá trên LLM hay dữ liệu doanh nghiệp thật.
    \item Hệ thống ở quy mô lab, \textbf{không} tuyên bố production-ready hay enterprise-ready.
\end{itemize}

\section{Kết luận chương}

Kiểm thử lab cho thấy tường guard nhiều lớp phát hiện $77\%$ đòn tấn công tổng
hợp ở FPR $0\%$ trên bộ đánh giá, với phân tích trung thực về giới hạn recall,
precision và phép đo rò rỉ. Mọi diễn giải nằm trong giới hạn bằng chứng tổng hợp
và mô hình cục bộ.
